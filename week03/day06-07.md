"""
===== MyAgent — 多工具 AI Agent =====

功能定位：
  一个不依赖任何 Agent 框架（LangChain / CrewAI / AutoGen）、
  纯手写的多工具 Agent。用 httpx 直接调 LLM API，
  手写 Agent Loop，手写工具调度。

工具列表（6 个）：
  1. get_weather     — 天气查询（模拟）
  2. calculate       — 数学计算（安全沙箱）
  3. get_current_time — 日期时间
  4. search_web      — 网络搜索（DuckDuckGo）
  5. save_note       — 保存笔记到内存知识库
  6. get_note        — 检索笔记

核心设计原则：
  - messages 是唯一的状态源，每轮只追加不修改
  - 工具执行结果用 tool 角色返回，不混入 user/assistant
  - 错误不抛异常，而是作为 tool 结果发给 LLM 自行处理
  - 所有工具都是幂等的（多次调用不产生副作用）

输出物：
  week03/day07/my_agent.py  ← 最终可运行的 Agent
"""

```python
"""day07/my_agent.py — 完整多工具 Agent"""

import json
import httpx
import os
import re
import random
from datetime import datetime
from typing import Callable


class MyAgent:
    """
    完整的多工具 Agent 实现。

    设计要点：
    - __init__ 中注册所有工具
    - register_tool() 对外开放，方便扩展
    - run() 是核心 Agent Loop（非流式）
    - chat() 提供交互式对话模式
    - 所有统计归入 reset_stats()
    """

    def __init__(
        self,
        system_prompt: str | None = None,
        model: str = "gpt-4o",
        api_key: str = "",
        base_url: str = "https://api.openai.com/v1",
        max_turns: int = 15,
        max_tokens_per_turn: int = 4096,
    ):
        self.model = model
        self.api_key = api_key or os.getenv("OPENAI_API_KEY", "")
        self.base_url = base_url
        self.max_turns = max_turns
        self.max_tokens_per_turn = max_tokens_per_turn

        # 注册工具
        self.tools: list[dict] = []
        self.handlers: dict[str, Callable] = {}
        self._register_default_tools()

        # 系统提示词
        self.system_prompt = system_prompt or """你是一个功能强大的 AI 助手。
你有多种工具可以使用，请根据用户需求选择合适的工具。
如果工具返回错误，请尝试其他方法或告诉用户。
用中文回答。"""

        # 统计
        self.reset_stats()

    def reset_stats(self):
        """重置所有统计指标"""
        self.total_turns = 0
        self.total_input_tokens = 0
        self.total_output_tokens = 0
        self.total_cost = 0.0
        self.tool_call_count = 0
        self._notes = {}  # 知识库（每次重置清空）

    # ==========================================
    # 工具注册系统
    # ==========================================

    def register_tool(self, schema: dict, handler: Callable):
        """
        注册一个新工具。

        参数：
          schema  — OpenAI 格式的工具定义（含 name, description, parameters）
          handler — 实际执行的 Python 函数

        示例：
          agent.register_tool(TOOL_SCHEMA, my_handler)
        """
        self.tools.append(schema)
        name = schema["function"]["name"]
        self.handlers[name] = handler

    def _register_default_tools(self):
        """注册默认的 6 个工具"""

        # ---------- 1. 天气查询 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "get_weather",
                    "description": "获取指定城市的实时天气。输入城市名，返回温度、天气状况、湿度。",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "city": {
                                "type": "string",
                                "description": "城市名，如 北京、上海、广州、Tokyo",
                            },
                        },
                        "required": ["city"],
                    },
                },
            },
            self._handle_weather,
        )

        # ---------- 2. 数学计算 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "calculate",
                    "description": "执行数学计算。支持加减乘除、括号、幂运算。例如：1024 * 768, (15 + 25) * 2, 2 ** 10",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "expression": {
                                "type": "string",
                                "description": "数学表达式，如 '1024 * 768'",
                            },
                        },
                        "required": ["expression"],
                    },
                },
            },
            self._handle_calculate,
        )

        # ---------- 3. 当前时间 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "get_current_time",
                    "description": "获取当前日期和时间。不依赖任何外部 API。当用户问\"现在几点\"、\"今天几号\"、\"星期几\"时调用。",
                    "parameters": {"type": "object", "properties": {}},
                },
            },
            self._handle_time,
        )

        # ---------- 4. 网络搜索 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "search_web",
                    "description": "搜索互联网获取最新信息。当需要实时数据、新闻、或你不知道的知识时调用。使用 DuckDuckGo 搜索引擎，无需 API key。",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "query": {
                                "type": "string",
                                "description": "搜索关键词，越具体越好。例如：'2026年 AI Agent 市场规模'",
                            },
                            "max_results": {
                                "type": "integer",
                                "description": "返回结果数，1-5 条",
                                "default": 3,
                            },
                        },
                        "required": ["query"],
                    },
                },
            },
            self._handle_search,
        )

        # ---------- 5. 保存笔记 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "save_note",
                    "description": "保存一条笔记到知识库。当用户说\"记住\"或\"记下来\"时调用。笔记会被存储并在后续对话中可检索。",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "key": {
                                "type": "string",
                                "description": "笔记的键，用于检索。例如：user_preference_color、meeting_notes_20260101",
                            },
                            "value": {
                                "type": "string",
                                "description": "笔记内容",
                            },
                        },
                        "required": ["key", "value"],
                    },
                },
            },
            self._handle_save_note,
        )

        # ---------- 6. 检索笔记 ----------
        self.register_tool(
            {
                "type": "function",
                "function": {
                    "name": "get_note",
                    "description": "从知识库中检索笔记。当用户说\"我之前说过\"、\"还记得吗\"、\"查一下我记的\"时调用。",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "key": {
                                "type": "string",
                                "description": "要检索的笔记键名",
                            },
                        },
                        "required": ["key"],
                    },
                },
            },
            self._handle_get_note,
        )

    # ==========================================
    # 工具 Handler 实现
    # ==========================================

    def _handle_weather(self, city: str) -> dict:
        """模拟天气查询（可替换为真实 API）"""
        conditions = ["晴 ☀️", "多云 ⛅", "阴 ☁️", "小雨 🌦️", "晴转多云 🌤️"]
        return {
            "city": city,
            "temperature": random.randint(15, 35),
            "condition": random.choice(conditions),
            "humidity": random.randint(30, 80),
            "wind": f"{random.randint(1, 5)} 级",
            "query_time": datetime.now().strftime("%H:%M"),
        }

    def _handle_calculate(self, expression: str) -> dict:
        """安全数学计算——带字符白名单校验"""
        # 安全检查：只允许数字、运算符、小数点、空格
        allowed = set("0123456789+-*/.()% ")
        if not all(c in allowed for c in expression):
            return {
                "error": "表达式包含非法字符",
                "expression": expression,
                "allowed_chars": "0123456789+-*/.()% 空格",
            }
        # 禁止字母（防止 __import__、eval 注入）
        if re.search(r"[a-zA-Z]", expression):
            return {
                "error": "表达式中不允许使用字母",
                "expression": expression,
            }
        try:
            # ⚠️ 生产环境请用 numexpr 替代 eval
            result = eval(expression)
            return {"expression": expression, "result": result}
        except Exception as e:
            return {
                "error": f"计算错误: {str(e)}",
                "expression": expression,
            }

    def _handle_time(self) -> dict:
        """获取当前时间"""
        now = datetime.now()
        return {
            "datetime": now.isoformat(),
            "date": now.strftime("%Y-%m-%d"),
            "time": now.strftime("%H:%M:%S"),
            "weekday": ["周一", "周二", "周三", "周四", "周五", "周六", "周日"][
                now.weekday()
            ],
            "timezone": "Asia/Shanghai (UTC+8)",
        }

    def _handle_search(self, query: str, max_results: int = 3) -> dict:
        """
        搜索互联网（使用 DuckDuckGo Lite HTML API，无需 API key）
        """
        try:
            # 限制最大结果数
            max_results = min(max(1, max_results), 5)

            resp = httpx.get(
                "https://html.duckduckgo.com/html/",
                params={"q": query},
                headers={
                    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                    "AppleWebKit/537.36 (KHTML, like Gecko) "
                    "Chrome/120.0.0.0 Safari/537.36"
                },
                timeout=10,
            )

            if resp.status_code != 200:
                return {
                    "query": query,
                    "results": [],
                    "error": f"搜索服务返回状态码 {resp.status_code}",
                }

            # 简易 HTML 解析提取搜索结果
            results = []
            # 用正则分割提取每个结果块
            blocks = resp.text.split('<a rel="nofollow" href="')
            for block in blocks[1 : max_results + 1]:
                url = block.split('"')[0]
                # 提取标题
                title_match = re.search(
                    r'class="result__title".*?>(.*?)</a>', block, re.DOTALL
                )
                # 提取摘要
                snippet_match = re.search(
                    r'class="result__snippet".*?>(.*?)</', block, re.DOTALL
                )

                title = (
                    re.sub(r"<[^>]+>", "", title_match.group(1)).strip()
                    if title_match
                    else url
                )
                snippet = (
                    re.sub(r"<[^>]+>", "", snippet_match.group(1)).strip()
                    if snippet_match
                    else ""
                )

                results.append(
                    {"title": title, "url": url, "snippet": snippet}
                )

            return {
                "query": query,
                "results": results,
                "total": len(results),
            }

        except httpx.TimeoutException:
            return {
                "query": query,
                "results": [],
                "error": "搜索请求超时，请稍后重试",
            }
        except Exception as e:
            return {
                "query": query,
                "results": [],
                "error": f"搜索失败: {str(e)}",
            }

    def _handle_save_note(self, key: str, value: str) -> dict:
        """保存笔记到内存（后续可改为 Chroma / 文件持久化）"""
        self._notes[key] = value
        return {
            "status": "saved",
            "key": key,
            "note_count": len(self._notes),
        }

    def _handle_get_note(self, key: str) -> dict:
        """检索笔记"""
        value = self._notes.get(key)
        if value is not None:
            return {"key": key, "value": value, "found": True}
        # 没找到时，列出相近的 key 帮助 LLM
        similar = [k for k in self._notes if key in k or k in key]
        return {
            "key": key,
            "found": False,
            "message": f"未找到键为 '{key}' 的笔记",
            "available_keys": similar if similar else list(self._notes.keys()),
        }

    # ==========================================
    # LLM 调用封装
    # ==========================================

    def _call_llm(self, messages: list[dict]) -> dict:
        """调用 LLM API（OpenAI 兼容格式）"""
        if not self.api_key:
            raise ValueError(
                "API key 未设置！请通过环境变量 OPENAI_API_KEY 或初始化参数设置。"
            )

        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }
        body = {
            "model": self.model,
            "messages": messages,
            "tools": self.tools,
            "tool_choice": "auto",
            "max_tokens": self.max_tokens_per_turn,
            "temperature": 0.7,
        }

        with httpx.Client(timeout=120) as client:
            resp = client.post(
                f"{self.base_url}/chat/completions",
                headers=headers,
                json=body,
            )
            resp.raise_for_status()
            data = resp.json()

            # Token 统计
            if "usage" in data:
                self.total_input_tokens += data["usage"].get("prompt_tokens", 0)
                self.total_output_tokens += data["usage"].get(
                    "completion_tokens", 0
                )

            return data

    # ==========================================
    # 核心 Agent Loop
    # ==========================================

    def run(self, user_input: str, verbose: bool = True) -> str:
        """
        主入口：接受用户输入，返回 Agent 回答。

        参数：
          user_input — 用户输入文本
          verbose    — 是否打印中间过程（默认 True）

        返回：
          Agent 的最终回答文本
        """
        messages = [
            {"role": "system", "content": self.system_prompt},
            {"role": "user", "content": user_input},
        ]

        for turn in range(self.max_turns):
            self.total_turns += 1

            if verbose:
                print(f"\n── Turn {turn+1}/{self.max_turns} ──")

            # 1. 调用 LLM
            response = self._call_llm(messages)
            assistant_msg = response["choices"][0]["message"]

            # 2. 检查是否有 tool_calls
            if not assistant_msg.get("tool_calls"):
                # 无 tool_calls → 最终回答
                final_text = assistant_msg.get("content", "") or ""
                if verbose:
                    print(f"📥 {final_text[:300]}")
                return final_text

            # 3. 有 tool_calls → 追加 assistant 消息
            messages.append(assistant_msg)

            # 4. 逐个执行工具
            for tc in assistant_msg["tool_calls"]:
                func_name = tc["function"]["name"]
                func_args = json.loads(tc["function"]["arguments"])
                self.tool_call_count += 1

                if verbose:
                    args_str = json.dumps(func_args, ensure_ascii=False)[:80]
                    print(f"  🔧 {func_name}({args_str})")

                # 查找 handler 并执行
                handler = self.handlers.get(func_name)
                if handler:
                    try:
                        result = handler(**func_args)
                    except TypeError as e:
                        # 参数不匹配（LLM 传了多余参数）
                        result = {
                            "error": f"参数错误: {str(e)}",
                            "expected_args": list(
                                handler.__code__.co_varnames[
                                    1 : handler.__code__.co_argcount
                                ]
                            ),
                        }
                    except Exception as e:
                        result = {"error": f"工具执行失败: {str(e)}"}
                else:
                    result = {
                        "error": f"未知工具: {func_name}",
                        "available_tools": list(self.handlers.keys()),
                    }

                # 5. 将工具结果追加到 messages
                messages.append(
                    {
                        "role": "tool",
                        "tool_call_id": tc["id"],
                        "content": json.dumps(result, ensure_ascii=False),
                    }
                )

        # 达到最大轮数
        return f"⚠️ 已达到最大轮数限制 ({self.max_turns})，Agent 已停止。请简化你的问题或拆分提问。"

    # ==========================================
    # 交互式对话模式
    # ==========================================

    def chat(self):
        """
        交互式对话模式。

        在终端中启动一个 REPL 循环：
        - 输入内容 → Agent 处理 → 输出结果
        - 输入 /quit 或 /exit 退出
        - 退出时打印会话统计
        """
        print("=" * 56)
        print("  🤖 MyAgent v1.0 — 多工具 AI Agent")
        print("  输入 /quit 退出，输入 /stats 查看统计")
        print(f"  模型: {self.model}")
        print(f"  工具: {len(self.tools)} 个")
        print("=" * 56)

        while True:
            try:
                user_input = input("\n👤 ").strip()
            except (EOFError, KeyboardInterrupt):
                print()  # 换行
                break

            if user_input.lower() in ("/quit", "/exit", "/q"):
                break
            if user_input.lower() in ("/stats", "/status"):
                self._print_stats()
                continue
            if not user_input:
                continue

            try:
                result = self.run(user_input)
                print(f"\n🤖 {result}")
            except Exception as e:
                print(f"\n❌ 错误: {str(e)}")
                print("请检查 API key 和网络连接。")

        self._print_stats()

    def _print_stats(self):
        """打印当前会话统计"""
        print(f"\n{'=' * 56}")
        print("  📊 会话统计")
        print(f"  Turn 总数:    {self.total_turns}")
        print(f"  工具调用数:   {self.tool_call_count}")
        print(f"  输入 Token:   {self.total_input_tokens:,}")
        print(f"  输出 Token:   {self.total_output_tokens:,}")
        if self.total_input_tokens or self.total_output_tokens:
            total_tokens = self.total_input_tokens + self.total_output_tokens
            print(f"  总 Token 数:  {total_tokens:,}")
        print("=" * 56)


# ==========================================
# 启动入口
# ==========================================

if __name__ == "__main__":
    # 创建 Agent 实例并启动交互模式
    agent = MyAgent(
        model="gpt-4o",
        # 你也可以换成其他模型：
        # model="claude-sonnet-4-20250514",
        # model="deepseek-chat",
        # base_url="https://api.deepseek.com/v1",
    )
    agent.chat()
```

## 三、交互式对话模式详解

上面的 `chat()` 方法实现了完整的交互式对话。我们来拆解它做了什么：

### 3.1 流程

```
用户输入
  │
  ▼
  ├─ /quit ──→ 退出，打印统计
  ├─ /stats ─→ 打印统计，继续
  ├─ 空输入 ─→ 忽略，继续
  │
  └─ 正常输入
       │
       ▼
      run(user_input)   ← 进入 Agent Loop（上面第 2 节）
       │
       ▼
   打印结果 ←────────── 回到等待输入
```

## 踩坑记录（Day 06）

### 坑 1：API key 明明设了，程序说找不到
- **症状**：`ValueError: API key 未设置`
- **原因**：`os.getenv("OPENAI_API_KEY")` 读的是环境变量，不是 `.env` 文件
- **解决**：要么 `export OPENAI_API_KEY=...`，要么用 `python-dotenv` 加载 `.env`
  ```python
  from dotenv import load_dotenv
  load_dotenv()  # 在 __init__ 最前面加
  ```

### 坑 2：LLM 传了 handler 不接受的参数
- **症状**：`TypeError: _handle_weather() got an unexpected keyword argument 'unit'`
- **原因**：LLM 根据 description 推断出多余的参数（比如它以为天气查询需要 unit 参数）
- **解决**：handler 用 `**kwargs` 吸收多余参数：
  ```python
  def _handle_weather(self, city: str, **kwargs) -> dict:
  ```

### 坑 3：eval 的安全问题
- **症状**：一切正常，但 `eval()` 是著名的安全漏洞
- **解决**：
  - 白名单字符校验（上面的代码已经做了）
  - 生产环境用 `numexpr` 库替代 `eval`
  - ```bash
    pip install numexpr
    # 然后：import numexpr; result = numexpr.evaluate(expression)
    ```

### 坑 4：messages 越来越长，Token 消耗暴增
- **症状**：多轮对话后，每次 API 调用的输入 token 数翻倍
- **原因**：Agent Loop 不断追加消息，从不清理
- **解决方向**：
  - 设置 `max_turns` 上限（已做）
  - 实现上下文窗口管理：当 messages 超过一定长度后，截断早期的 tool 结果
  - 使用 Prompt Caching（OpenAI / Anthropic 都支持）

### 坑 5：百度/搜狗等中文搜索请求被 DuckDuckGo 拒绝
- **症状**：搜索中国相关内容返回空结果
- **原因**：DuckDuckGo 对中国 IP 的限制
- **解决**：fallback 到 Bing Search API（需要申请 key，Azure 免费层可用）

### 坑 6：Agent 陷入「工具循环」
- **症状**：LLM 反复调用同一个工具（比如连续 5 次 `get_weather("北京")`），每次收到相同结果仍继续调用
- **原因**：messages 上下文太长，LLM "忘了"自己已经查过
- **解决**：增加去重检测——如果连续两次相同 tool_call，强制终止
  ```python
  # 在 run() 里加检测
  last_call = None
  for tc in assistant_msg["tool_calls"]:
      current_call = (tc["function"]["name"], tc["function"]["arguments"])
      if current_call == last_call:
          if verbose:
              print("  ⚠️ 检测到重复 tool call，中断循环")
          break
      last_call = current_call
  ```

### 挑战 1：并行工具调用 🔁

当前 `run()` 方法是**串行**执行 tool_calls 的：等第一个执行完，再执行第二个。如果 LLM 一次返回了 3 个工具调用（比如查天气 + 查时间 + 搜索），串行效率很低。

```python
"""并行工具调用示例"""
import asyncio
import json


async def run_async(self, user_input: str) -> str:
    """异步版 run()——并行执行 tool_calls"""
    messages = [
        {"role": "system", "content": self.system_prompt},
        {"role": "user", "content": user_input},
    ]

    for turn in range(self.max_turns):
        # LLM 调用（仍然是同步的——API 调用没法并行）
        response = self._call_llm(messages)
        assistant_msg = response["choices"][0]["message"]

        if not assistant_msg.get("tool_calls"):
            return assistant_msg.get("content", "") or ""

        messages.append(assistant_msg)

        # ⚡ 并行执行所有 tool_calls
        async def execute_one(tc):
            func_name = tc["function"]["name"]
            func_args = json.loads(tc["function"]["arguments"])
            handler = self.handlers.get(func_name)
            if handler:
                result = handler(**func_args)
            else:
                result = {"error": f"未知工具: {func_name}"}
            return {
                "role": "tool",
                "tool_call_id": tc["id"],
                "content": json.dumps(result, ensure_ascii=False),
            }

        # 并发执行
        tool_results = await asyncio.gather(
            *[execute_one(tc) for tc in assistant_msg["tool_calls"]]
        )
        messages.extend(tool_results)

    return f"⚠️ 达到最大轮数 ({self.max_turns})"

# 注意：需要 async 入口

def tool_cache(maxsize: int = 128):

### 挑战 3：流式输出 ⚡

给 Agent 加上 streaming 模式，边生成边输出：

```python
"""流式版 MyAgent"""

def _call_llm_stream(self, messages: list[dict]):
    """流式调用 LLM API"""
    headers = {
        "Authorization": f"Bearer {self.api_key}",
        "Content-Type": "application/json",
    }
    body = {
        "model": self.model,
        "messages": messages,
        "tools": self.tools,
        "tool_choice": "auto",
        "stream": True,  # ⚡ 关键
    }

    with httpx.Client(timeout=120) as client:
        with client.stream(
            "POST",
            f"{self.base_url}/chat/completions",
            headers=headers,
            json=body,
        ) as resp:
            for line in resp.iter_lines():
                if line.startswith("data: "):
                    data_str = line[6:]
                    if data_str == "[DONE]":
                        break
                    try:
                        chunk = json.loads(data_str)
                        yield chunk
                    except json.JSONDecodeError:
                        continue


def run_stream(self, user_input: str):
    """
    流式版 run()。

    先流式输出思考过程，需要调用工具时暂停并执行，
    然后用流式输出最终回答。
    """
    messages = [
        {"role": "system", "content": self.system_prompt},
        {"role": "user", "content": user_input},
    ]

    for turn in range(self.max_turns):
        collected_content = ""
        tool_calls_buffer = {}

        print(f"\n── Turn {turn+1} ──")

        for chunk in self._call_llm_stream(messages):
            delta = chunk.get("choices", [{}])[0].get("delta", {})

            # 文本内容
            if "content" in delta and delta["content"]:
                content = delta["content"]
                collected_content += content
                print(content, end="", flush=True)

            # tool_calls
            if "tool_calls" in delta:
                for tc in delta["tool_calls"]:
                    idx = tc.get("index", 0)
                    if idx not in tool_calls_buffer:
                        tool_calls_buffer[idx] = {
                            "id": tc.get("id", ""),
                            "function": {
                                "name": tc.get("function", {}).get("name", ""),
                                "arguments": "",
                            },
                        }
                    if "function" in tc and "arguments" in tc["function"]:
                        tool_calls_buffer[idx]["function"]["arguments"] += (
                            tc["function"]["arguments"]
                        )

        print()  # 换行

        # 如果有 tool_calls，执行
        assistant_msg = {"content": collected_content or None}
        if tool_calls_buffer:
            tc_list = []
            for idx in sorted(tool_calls_buffer.keys()):
                tc = tool_calls_buffer[idx]
                tc_list.append({
                    "id": tc["id"],
                    "type": "function",
                    "function": tc["function"],
                })
            assistant_msg["tool_calls"] = tc_list

        if not assistant_msg.get("tool_calls"):
            return collected_content

        messages.append(assistant_msg)
        # ... 执行工具逻辑同上 ...

## 踩坑记录（Day 07）

### 坑 1：测试不联网就跑不了
- **问题**：所有测试都依赖真实 API 调用
- **解法**：写 mock 版 LLM 返回

```python
# test_agent_mock.py — 不依赖 API 的测试
from my_agent import MyAgent


class MockAgent(MyAgent):
    """Mock 版 Agent——不调真实 LLM，直接返回预设 tool_calls"""

    def _call_llm(self, messages):
        """返回预设响应，不调 API"""
        last_msg = messages[-1]
        if last_msg["role"] == "user" and "计算" in last_msg["content"]:
            return {
                "choices": [{
                    "message": {
                        "role": "assistant",
                        "content": None,
                        "tool_calls": [{
                            "id": "call_mock_1",
                            "type": "function",
                            "function": {
                                "name": "calculate",
                                "arguments": '{"expression": "1024*768"}',
                            },
                        }],
                    }
                }]
            }
        if last_msg["role"] == "tool":
            return {
                "choices": [{
                    "message": {
                        "role": "assistant",
                        "content": f"计算结果：{json.loads(last_msg['content'])['result']}",
                    }
                }]
            }
        return {
            "choices": [{
                "message": {"role": "assistant", "content": "你好！"}
            }]
        }


def test_calculate_mock():
    agent = MockAgent(api_key="mock")
    result = agent.run("1024 * 768 等于多少？", verbose=False)
    assert "786432" in result
    print("✅ Mock 测试通过（无 API 调用）")
```

### 坑 2：连续两次 `run()` 之间的状态共享
- **问题**：`chat()` 模式下，每次 `run()` 是独立的（不共享对话历史），但 `_notes` 是共享的
- **原因**：`run()` 每次重新构建 `messages`，只包含 system + user，不包含历史
- **影响**：Agent 记不住上一轮说了什么（除非用 `chat()` 模式）
- **修复方向**：`chat()` 方法中把 `messages` 提到循环外，跨轮共享

### 坑 3：DuckDuckGo 搜索被限流
- **问题**：连续多次搜索后，DuckDuckGo 返回空结果
- **解决**：在搜索之间加 1 秒延迟，或 fallback 到备用搜索源
